cmake_minimum_required(VERSION 3.24)
project(verilog2digital_gui VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.2 REQUIRED CONFIG COMPONENTS core base)
find_package(nlohmann_json 3.12 REQUIRED CONFIG)
add_library(v2d-desktop STATIC src/worker_bridge.cpp src/preview_panel.cpp src/project_document.cpp src/main_frame.cpp)
target_include_directories(v2d-desktop PUBLIC src)
target_link_libraries(v2d-desktop PUBLIC wx::core wx::base nlohmann_json::nlohmann_json bcrypt shell32)
target_compile_definitions(v2d-desktop PUBLIC UNICODE _UNICODE)
add_executable(v2d-gui WIN32 src/main.cpp src/application.rc)
target_link_libraries(v2d-gui PRIVATE v2d-desktop)
add_executable(VerilogToDigital WIN32 src/portable_launcher.cpp src/application.rc)
target_link_libraries(VerilogToDigital PRIVATE nlohmann_json::nlohmann_json user32 shell32)
target_compile_definitions(VerilogToDigital PRIVATE UNICODE _UNICODE)
target_link_options(VerilogToDigital PRIVATE -municode -static -Wl,--no-insert-timestamp)
include(CTest)
if(BUILD_TESTING)
  add_executable(v2d-preview-tests tests/preview_panel_test.cpp)
  target_link_libraries(v2d-preview-tests PRIVATE v2d-desktop)
  add_executable(v2d-bridge-tests tests/worker_bridge_test.cpp)
  target_link_libraries(v2d-bridge-tests PRIVATE v2d-desktop)
  add_executable(v2d-project-tests tests/project_document_test.cpp)
  target_link_libraries(v2d-project-tests PRIVATE v2d-desktop)
  add_executable(v2d-main-frame-tests tests/main_frame_test.cpp src/application.rc)
  target_link_libraries(v2d-main-frame-tests PRIVATE v2d-desktop)
endif()
