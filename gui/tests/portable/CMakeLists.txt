cmake_minimum_required(VERSION 3.24)
project(v2d_portable_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(wxWidgets 3.2 REQUIRED CONFIG COMPONENTS core base)
add_executable(VerilogToDigital WIN32 ../../src/portable_launcher.cpp)
target_link_libraries(VerilogToDigital PRIVATE nlohmann_json::nlohmann_json user32 shell32)
target_link_options(VerilogToDigital PRIVATE -municode -static -static-libgcc -static-libstdc++)
add_executable(v2d-portable-tests ../portable_launcher_test.cpp)
target_link_libraries(v2d-portable-tests PRIVATE nlohmann_json::nlohmann_json user32 shell32)
target_link_options(v2d-portable-tests PRIVATE -static -static-libgcc -static-libstdc++)
add_executable(v2d-portable-bridge-tests ../worker_bridge_test.cpp ../../src/worker_bridge.cpp)
target_include_directories(v2d-portable-bridge-tests PRIVATE ../../src)
target_link_libraries(v2d-portable-bridge-tests PRIVATE wx::core wx::base nlohmann_json::nlohmann_json shell32)
