cmake_minimum_required(VERSION 3.24)
project(v2d_preview_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(wxWidgets 3.2 REQUIRED CONFIG COMPONENTS core base)
find_package(nlohmann_json CONFIG REQUIRED)
add_executable(v2d-preview-test ../preview_panel_test.cpp ../../src/preview_panel.cpp)
target_include_directories(v2d-preview-test PRIVATE ../../src)
target_link_libraries(v2d-preview-test PRIVATE wx::core wx::base nlohmann_json::nlohmann_json bcrypt)
add_executable(v2d-project-test ../project_document_test.cpp ../../src/project_document.cpp)
target_include_directories(v2d-project-test PRIVATE ../../src)
target_link_libraries(v2d-project-test PRIVATE wx::base nlohmann_json::nlohmann_json bcrypt)
